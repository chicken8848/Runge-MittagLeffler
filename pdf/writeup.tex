\section{Runge's Theorem}
The fact that functions analytic on a disk are limits of polynomials is due to the fact that the disks are simply connected.
From the Laurent series, we can see that each analytic function on $G$ is the uniform limit of rational functions whose poles
lie outside $G$. Each holomorphic function $f$ is the limit of a sequence of rational functions which also belong to the functions
holomorphic on $G$. For example, for any Laurent series on an annulus,
\[
\begin{aligned}
f(z) = \underbrace{\sum_{n=0}^{\infty}c_n(z-a)^n}_{\text{Taylor Part}} + \underbrace{\sum_{n=0}^{\infty}\frac{c_{-n}}{(z-a)^n}}_{\text{Principle Part}}
\end{aligned}
\]
by taking finite partial sums,
\[
\begin{aligned}
R_N(z) = \sum_{n=0}^{N}c_n(z-a)^n + \sum_{n=0}^{N}\frac{c_{-n}}{(z-a)^n}
\end{aligned}
\]
We see that $R_N$ is a rational function, and $f$ is the limit of $R_N$.
This may be generalized to arbitrary regions, which is part of the content of Runge's Theorem. But first we need
the enhanced separation lemma.

\subsection{Enhanced Separation Lemma}
Let $G \subset \mathbb{C}$ be an open set, and let $K \subset G$ be a compact subset. Then there are straight line segments
$\gamma_1, \dots, \gamma_n$ in $G - K$ such that for every function $f$ that is holomorphic on $G$,
\[
\begin{aligned}
  f(z) = \sum_{k = 1}^{n} \frac{1}{2\pi i} \int_{\gamma_k} \frac{f(w)}{w-z}dw
\end{aligned}
\]
For all $z \in K$. The line segments form a finite number of closed polygons. Furthermore, also exists $\delta > 0$,
such that $\forall w \in \gamma_k,\forall k \in K, ||w - k|| \ge \delta$. In particular, we may find segments such that
$\forall k, ||\gamma_k(0) - \gamma_k(1)|| < 2 \delta$

\subsection{Rational approximation of a segment integral}
Let $K \subseteq \mathbb{C}$ and let $a,b,z_0\in\mathbb{C}$.
Suppose $\varphi:[a,b]\to\mathbb{C}$ is continuous and
there exist $r,\rho\in\mathbb{R}$ such that
\begin{enumerate}
    \item $0\leq r<\rho$;
    \item For every $t\in[0,1]$,
    \[
        |(1-t)a+tb-z_0|\leq r;
    \]
    \item For every $z\in K$,
    \[
        \rho<|z-z_0|.
    \]
\end{enumerate}
Then the function
\[
    F(z)=\int_{[a,b]}\frac{\varphi(w)}{w-z}\,dw
\]
can be uniformly approximated on $K$ by rational functions
whose only possible pole is $z_0$.

\begin{proof}
Since $\varphi$ is continuous on the compact segment $[a,b]$,
there exists $M\geq 0$ such that
\[
    |\varphi(w)|\leq M,
    \qquad w\in[a,b].
\]
Since $0\leq r<\rho$, we have $\rho>0$ and
\[
    0\leq\frac{r}{\rho}<1.
\]
Let $\varepsilon>0$. By the convergence of the geometric
sequence, there exists $N\in\mathbb{N}$ such that
\[
    \left(\frac{r}{\rho}\right)^{N+1}
    \frac{M|b-a|}{\rho-r}<\varepsilon.
\]
For $w\in[a,b]$ and $z\in K$, we have
\[
    \left|\frac{w-z_0}{z-z_0}\right|
    \leq\frac{r}{\rho}<1.
\]
Hence we may use the geometric expansion
\[
    \frac{1}{w-z}
    =-\frac{1}{z-z_0}
    \sum_{k=0}^{N}
    \left(\frac{w-z_0}{z-z_0}\right)^k
    +E_N(w,z),
\]
where
\[
    E_N(w,z)
    =-\frac{1}{z-z_0}
    \frac{\left(\frac{w-z_0}{z-z_0}\right)^{N+1}}
         {1-\frac{w-z_0}{z-z_0}}.
\]
Define
\[
    c_k=\int_{[a,b]}\varphi(w)(w-z_0)^k\,dw
\]
and
\[
    R_N(z)
    =-\sum_{k=0}^{N}\frac{c_k}{(z-z_0)^{k+1}}.
\]
Clearly, $R_N$ is a rational function whose only possible
pole is $z_0$.
Integrating the finite geometric expansion gives
\[
    F(z)-R_N(z)
    =\int_{[a,b]}\varphi(w)E_N(w,z)\,dw.
\]
Moreover, since $|w-z_0|\leq r$ and $|z-z_0|>\rho$,
\begin{align*}
    |E_N(w,z)|
    &=\frac{|w-z_0|^{N+1}}
    {|z-z_0|^{N+1}|w-z|}\\
    &\leq
    \left(\frac{r}{\rho}\right)^{N+1}
    \frac{1}{\rho-r},
\end{align*}
where we used
\[
    |w-z|
    \geq |z-z_0|-|w-z_0|
    >\rho-r.
\]
Consequently, by the integral estimation inequality,
\begin{align*}
    |F(z)-R_N(z)|
    &\leq\int_{[a,b]}
       |\varphi(w)|\,|E_N(w,z)|\,|dw|\\
    &\leq
    M|b-a|
    \left(\frac{r}{\rho}\right)^{N+1}
    \frac{1}{\rho-r}\\
    &<\varepsilon.
\end{align*}
This estimate holds for every $z\in K$. Therefore,
\[
    \sup_{z\in K}|F(z)-R_N(z)|
    \leq
    M|b-a|
    \left(\frac{r}{\rho}\right)^{N+1}
    \frac{1}{\rho-r}
    <\varepsilon.
\]
Since $\varepsilon>0$ was arbitrary, $F$ can be
uniformly approximated on $K$ by rational functions
whose only possible pole is $z_0$.
\end{proof}

\subsection{Weak Runge}
Let $G \subset \mathbb{C}$ be an open set, and let $K \subset G$ be a compact subset. 
Then for any holomorphic function $f: G \to \mathbb{C}$ on $G$ and any $\varepsilon > 0$, there exists a rational function $R(z)$
whose poles lie in $\mathbb{C}_{\infty} - K$
such that $|f(z) - R(z)| < \varepsilon$, for all $z \in K$.
\begin{proof}
  Since we have $G$ as an open set, with $K \subset G$ as a compact subset, and $f$ holomorphic on $G$,
  we apply the enhanced separation lemma to obtain,
  \[
  \begin{aligned}
  f(z) &= \sum_{i = 1}^{n} \frac{1}{2\pi i} \int_{\gamma_i} \frac{f(w)}{w-z} \,dw && \just{Enhanced separation lemma}
  \end{aligned}
  \]
  For all $i \in 1,...,n$, where $\gamma_i \in G - K$ is a straight line segment and exists $\delta > 0$,
  such that $\forall w \in \gamma_i,\forall k \in K, ||w - k|| \ge \delta$. Now in the above formulation of $f(z)$, let
  \[
  f_i = \frac{1}{2\pi i} \int_{\gamma_i} \frac{f(w)}{w-z} \,dw
  \]
  Then
  \[
  \begin{aligned}
    f(z) = \sum_{i = 1}^{n} f_i && \just{Substitution}
  \end{aligned}
  \]
  Now we claim that $f_i$ may be approximated on $K$ by rational functions with poles outside of $K$.
  For arbitrary $k$, let $a = \gamma_i(0)$ and $b = \gamma_i(1)$, then
  \begin{enumerate}
  \item Since $f$ is continuous on $G$, it is continuous on $G - K$, since all $\gamma_k \subset G - K$,
  we have that $f$ is continuous on $\gamma_k$.
  \item Every point on the segment is at most half the segment's length to the midpoint.
  \item From the enhanced separation lemma, we have $\frac{1}{2}||b - a||  < \delta$
  \item $\forall k \in K, \forall t \in [0, 1], \quad \delta \le ||(1-t)\gamma_i(0) + t\gamma_i(1) - k||$
    \just{Enhanced separation lemma}
  \item $\forall k \in K, \frac{\frac{||b - a||}{2} + \delta}{2} < \delta \le ||k - \frac{a + b}{2}||$
  \end{enumerate}
  Then by the rational approximation of a segment integral, each $f_i$ can be approximated on $K$
  by rational functions whose pole has some distance away from all $k \in K$. More precisely, 
  $\forall \varepsilon > 0$ there exists a rational function $R_i$ such that
  \[
  |f_i(k) - R_i(k)| < \frac{\varepsilon}{n}, \quad \forall k \in K
  \]
  Let $R(z) = \sum_{i = 1}^{n} R_i(z)$, then
  \[
  \begin{aligned}
    |f(z) - R(z)| &= | \sum_{i = 1}^{n}f_i(z) - \sum_{i = 1}^{n} R_i(z)| \\
    &< \sum_{i=1}^{n}\frac{\varepsilon}{n} \\
    &= \varepsilon
  \end{aligned} 
  \]
\end{proof}
